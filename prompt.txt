rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // 投票新規作成バリデーション（count は必ず 1 から）
    function isValidVoteCreate() {
      return request.resource.data.keys().hasOnly(['count', 'updatedAt'])
        && request.resource.data.count == 1
        && request.resource.data.updatedAt is timestamp;
    }

    // 投票更新バリデーション（count は必ず +1 のみ）
    function isValidVoteUpdate() {
      return request.resource.data.keys().hasOnly(['count', 'updatedAt'])
        && request.resource.data.count == resource.data.count + 1
        && request.resource.data.updatedAt is timestamp;
    }

    // TiersList 票数
    match /bakatareCounts/{character} {
      allow read:   if true;
      allow create: if isValidVoteCreate();
      allow update: if isValidVoteUpdate();
      allow delete: if false;
    }

    // GenshinCheck 票数
    match /checkSheetVotes/{questionId}/characters/{charId} {
      allow read:   if true;
      allow create: if isValidVoteCreate();
      allow update: if isValidVoteUpdate();
      allow delete: if false;
    }

    // GenshinOmikuji ユーザーデータ＋占い統計（端末変更対応）
    match /omikujiUsers/{userId} {
      allow read, write: if true;
    }

    // その他すべて拒否
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
